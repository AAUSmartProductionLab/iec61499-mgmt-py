// 4diac FORTE for a Raspberry Pi (64-bit OS): same features plus GPIO via /dev/gpiochip.
// Build with: build-runtime.ps1 -Config configurations/pi   (downloads a cross compiler once)
include=${FBE_MAIN}/configurations/inc/default.txt
CMAKE_BUILD_TYPE=MinSizeRel
ARCH=aarch64-linux-musl
FBE_EXTERNAL_MODULES_DIR=${FBE_ROOT}/../.cache/export
// FORTE's local OPC UA handler serialises server access with its own mutex and releases it
// while a method call waits for RSP. With open62541's internal locking (UA_MULTITHREADING=100,
// the 1.5 default) a PUBLISH during a method call deadlocks; build open62541 single-threaded.
UA_MULTITHREADING:STRING=0
// POWERLINK is off in FORTE 3.x, but its dependency libpcap still builds and fails in the
// cross build (flex.exe: pipe failed).
DEPS=-openpowerlink
// musl gives threads 128 KiB of stack unless the binary's PT_GNU_STACK says more; open62541 in
// FORTE's OPC UA thread overflows that (segfault as soon as an OPC UA FB initialises).
// 8 MiB as glibc. MinSizeRel flags, because the toolchain file sets CMAKE_EXE_LINKER_FLAGS_INIT.
CMAKE_EXE_LINKER_FLAGS_MINSIZEREL:STRING=-Wl,-z,stack-size=8388608
